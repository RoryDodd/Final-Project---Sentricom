import bcrypt

def hash_api_key(plaintext: str):
    return bcrypt.hashpw(plaintext.encode(), bcrypt.gensalt()).decode()
    
def hash_password(plaintext: str):
    return bcrypt.hashpw(plaintext.encode(), bcrypt.gensalt()).decode()
    
if __name__ == "__main__":
    api_key = input("Enter API Key to hash: ")
    password = input("Enter password to hash: ") 
    
    print("\n--- HASHSED ---")
    print("API Key Hash:    ", hash_api_key(api_key))
    print("Password Hash:    ", hash_password(password))
